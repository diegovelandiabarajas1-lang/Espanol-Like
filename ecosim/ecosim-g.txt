# ECOSIM-G · Monte Carlo de economías en GPU, con C1EÑ
#
# Uso (desde D:\ProyectosJS, con el perfil C1EÑ):
#   node motor/util/n-node.js --c1en ecosim/ecosim-g.esl [economias] [ticks] [compradores] [semilla] [salida.csv|-] [empresasPorSector]
# Por defecto: 10 economías · 7200 ticks (2 h a un tick por segundo) · 2000 compradores por economía.
# En Kaggle, con las dos T4: 1000 economías, 500 por GPU (c1en/kaggle-ecosim-g.ipynb).
#
# LA IDEA: una economía es un «mundo». Mil economías con mil semillas distintas,
# avanzando a la vez, son mil historias posibles de la misma economía. Al final se
# pregunta en cuántas hubo crisis, cuántas empresas quebraron, cómo se repartió la
# riqueza. Eso es el Monte Carlo: no una corrida, sino la distribución.
#
# CÓMO ENTRA EN LA GPU: todo el estado vive en arreglos PLANOS de tamaño
# economías × N. El hilo h es el agente (h % N_AG) de la economía (h // N_AG), y
# cada fase del tick es un «paralelo» sobre todos los agentes de todas las
# economías a la vez. Nada se comparte entre economías: cada una lee y escribe
# solo su tramo. La bolsa —ordenar órdenes— es secuencial y va en la CPU cada
# minuto simulado; pesa poco porque son 60 ticks de GPU por una subasta.
#
# LAS TRES MASAS, POR ECONOMÍA: el dinero (centavos, enteros) solo se mueve;
# las acciones (ACC por empresa activa) se cancelan al quebrar; los bienes
# cumplen producido = consumido + inventario + destruido. Se verifica cada
# minuto, en todas las economías, con un «paralelo» que suma por economía.
#
# EL ESTANTE LIMITADO: cuando 3.000 agentes quieren 500 unidades, en GPU no hay
# «el primero que llega». Se cuenta la demanda con sumarAtomico y cada agente
# compra si azarHilo(...) < inventario / demanda: una lotería determinista que no
# depende del orden de los hilos y conserva los bienes (lo que se vende de más
# por redondeo de la lotería se devuelve con una segunda pasada que recorta).
#
# EL AZAR es azarHilo(semilla, hilo, canal), puro: la economía k usa la semilla
# base + k, así que cada mundo es distinto y todos reproducibles, en cualquier
# dispositivo, al bit.

# ─────────────────────────────────────────────── tick 0, en paralelo
fn iniciarAgente(h: entero, saldo, hambre, aburr, curio, empleo, s: entero, nAg: entero, nIni: entero) {
  fijo k = h // nAg
  fijo i = h % nAg
  saldo[h] = 100000
  hambre[h] = entero(piso(azarHilo(s + k, i, 2) * 4000))
  aburr[h] = entero(piso(azarHilo(s + k, i, 3) * 4000))
  curio[h] = entero(piso(azarHilo(s + k, i, 4) * 4000))
  empleo[h] = i % nIni
}
fn iniciarInversor(h: entero, saldoInv) { saldoInv[h] = 1600000 }
fn iniciarEmpresa(h: entero, activa, sector, caja, precio, inv, trab, precioAcc, calidad, estrellas, resenas, nacida, s: entero, nEmp: entero, nIni: entero, porSector: entero, nAg: entero) {
  fijo k = h // nEmp
  fijo e = h % nEmp
  si e < nIni {
    activa[h] = 1
    fijo sec = e // porSector
    sector[h] = sec
    caja[h] = 20000000
    var base = 800
    si sec == 1 { base = 1200 }
    si sec == 2 { base = 3000 }
    si sec == 3 { base = 1500 }
    si sec == 4 { base = 1000 }
    precio[h] = base + entero(piso(azarHilo(s + k, 1000 + e, 5) * (base // 2)))
    inv[h] = nAg * 4 // (5 * nIni) + 50
    trab[h] = nAg // nIni
    si e < nAg % nIni { trab[h] = trab[h] + 1 }
    precioAcc[h] = 10000
    calidad[h] = 1 + entero(piso(azarHilo(s + k, 1000 + e, 6) * 5))
    estrellas[h] = 20 * (1 + entero(piso(azarHilo(s + k, 1000 + e, 7) * 5))) * 100
    resenas[h] = 20
  } sino {
    activa[h] = 0
    resenas[h] = 1
  }
  nacida[h] = 0
}
# IPO: cada acción se asigna a un inversor al azar; el hilo es (economía, empresa, acción)
fn repartirAcciones(h: entero, tenencia, activa, s: entero, nEmp: entero, acc: entero, ten: entero, nInv: entero) {
  fijo ke = h // acc                 # índice plano de la empresa
  si activa[ke] == 1 {
    fijo k = ke // nEmp
    fijo a = h % acc
    fijo quien = 1 + entero(piso(azarHilo(s + k, 2000 + ke % nEmp, 8 + a) * nInv))
    sumarAtomico(tenencia, ke * ten + quien, 1)
  }
}

# ─────────────────────────────────────────────── fases del tick
# 1. cada agente decide qué quiere: barras, sector, dos candidatas, gana la mejor
fn elegir(h: entero, hambre, aburr, curio, eleccion, saldo, activa, precio, estrellas, resenas, sector, demanda, t: entero, s: entero, nAg: entero, nEmp: entero, porSector: entero) {
  fijo k = h // nAg
  fijo i = h % nAg
  var hh = hambre[h] + 3
  si hh > 10000 { hh = 10000 }
  hambre[h] = hh
  var aa = aburr[h] + 2
  si aa > 10000 { aa = 10000 }
  aburr[h] = aa
  var cc = curio[h] + 1
  si cc > 10000 { cc = 10000 }
  curio[h] = cc
  var sec = -1
  si hh >= 4000 y hh >= aa y hh >= cc { sec = (t + i) % 2 }
  sino si aa >= 4000 y aa >= cc { sec = 3 }
  sino si cc >= 4000 { sec = 2 }
  sino si (t + i * 7) % 3000 == 0 { sec = 4 }
  eleccion[h] = -1
  si sec >= 0 {
    # dos candidatas al azar entre las activas del sector (hasta 6 intentos)
    fijo r = azarHilo(s + k, h, 10 + t % 7)
    fijo base = k * nEmp
    var mejor = -1
    var mejorPz = 0
    var intento = 0
    mientras intento < 6 {
      fijo e = base + entero(piso(azarHilo(s + k, h, 20 + intento + 7 * (t % 11)) * nEmp))
      si activa[e] == 1 y sector[e] == sec {
        fijo pz = (estrellas[e] // resenas[e]) * 4 - precio[e]
        si mejor < 0 o pz > mejorPz {
          mejor = e
          mejorPz = pz
        }
      }
      intento = intento + 1
    }
    si mejor >= 0 {
      # lo que no es agua ni comida exige colchón
      si sec < 2 o saldo[h] >= precio[mejor] * 20 {
        eleccion[h] = mejor
        sumarAtomico(demanda, mejor, 1)
      }
    }
  }
}
# 2. lotería: si la demanda supera el inventario, compra una fracción; el dinero
#    y las unidades se mueven con atómicos
fn comprar(h: entero, eleccion, saldo, caja, inv, demanda, ventas, compro, precio, hambre, aburr, curio, sector, estrellas, resenas, calidad, consumido, t: entero, s: entero, nAg: entero) {
  compro[h] = 0
  fijo e = eleccion[h]
  si e >= 0 {
    fijo k = h // nAg
    fijo pr = precio[e]
    si saldo[h] >= pr {
      fijo d = demanda[e]
      fijo stock = inv[e]
      var puede = 1
      si d > stock {
        si azarHilo(s + k, h, 30 + t % 5) * d >= stock { puede = 0 }
      }
      si puede == 1 {
        saldo[h] = saldo[h] - pr
        sumarAtomico(caja, e, pr)
        sumarAtomico(ventas, e, 1)
        sumarAtomico(consumido, k, 1)
        compro[h] = 1
        fijo sec = sector[e]
        si sec <= 1 { hambre[h] = hambre[h] - 4500 }
        sino si sec == 3 { aburr[h] = aburr[h] - 4500 }
        sino si sec == 2 { curio[h] = curio[h] - 4500 }
        si hambre[h] < 0 { hambre[h] = 0 }
        si aburr[h] < 0 { aburr[h] = 0 }
        si curio[h] < 0 { curio[h] = 0 }
        si azarHilo(s + k, h, 40 + t % 3) < 0.1 {
          var est = calidad[e] + entero(piso(azarHilo(s + k, h, 50) * 3)) - 1
          si est < 1 { est = 1 }
          si est > 5 { est = 5 }
          sumarAtomico(estrellas, e, est * 100)
          sumarAtomico(resenas, e, 1)
        }
      }
    }
  }
}
# 2b. la lotería puede vender unas unidades de más (es probabilística): se
#     descuenta del inventario lo vendido; si queda negativo, las últimas compras
#     se deshacen con una pasada secuencial en la CPU (rara, y se contabiliza)
fn cerrarVentas(h: entero, inv, ventas, ventasHora, ingresosHora, precio, demanda, activa) {
  si activa[h] == 1 {
    inv[h] = inv[h] - ventas[h]
    ventasHora[h] = ventasHora[h] + ventas[h]
    ingresosHora[h] = ingresosHora[h] + ventas[h] * precio[h]
  }
  demanda[h] = 0
}
# 3. producir: 4,8 unidades por trabajador y hora
fn producir(h: entero, activa, progreso, trab, inv, producido, nEmp: entero) {
  si activa[h] == 1 {
    progreso[h] = progreso[h] + trab[h] * 48
    fijo u = progreso[h] // 36000
    progreso[h] = progreso[h] % 36000
    inv[h] = inv[h] + u
    sumarAtomico(producido, h // nEmp, u)
  }
}
# 4. cada hora: salarios. Un hilo por EMPRESA recorre a los agentes de su
#    economía y paga a los suyos mientras la caja alcance; al que no alcanza lo
#    despide. Secuencial dentro de la empresa, así el resultado no depende del
#    orden de los hilos (con un hilo por agente, quién cobra y quién no cuando la
#    caja se acaba lo decidiría la GPU).
fn pagarSalarios(h: entero, activa, empleo, saldo, caja, salariosHora, nAg: entero, nEmp: entero, salario: entero) {
  si activa[h] == 1 {
    fijo k = h // nEmp
    fijo e = h % nEmp
    fijo base = k * nAg
    para i en rango(nAg) {
      si empleo[base + i] == e {
        si caja[h] >= salario {
          caja[h] = caja[h] - salario
          saldo[base + i] = saldo[base + i] + salario
          salariosHora[h] = salariosHora[h] + salario
        } sino {
          empleo[base + i] = -1
        }
      }
    }
  }
}
# 5. cada hora: la empresa ajusta precio y plantilla, y mide su hora
fn cerrarHora(h: entero, activa, ventasHora, ingresosHora, salariosHora, ganHora, ganMax, horasMalas, inv, precio, trab, caja, nacida, t: entero, gracia: entero) {
  si activa[h] == 1 {
    var v: entero = ventasHora[h]
    si v < 1 { v = 1 }
    ganHora[h] = ingresosHora[h] - salariosHora[h]
    si ganHora[h] > ganMax[h] { ganMax[h] = ganHora[h] }
    si ingresosHora[h] < salariosHora[h] { precio[h] = precio[h] * 103 // 100 + 1 }
    sino si inv[h] > 2 * v {
      precio[h] = precio[h] * 98 // 100
      si precio[h] < 100 { precio[h] = 100 }
    }
    sino si inv[h] < v // 2 { precio[h] = precio[h] * 102 // 100 + 1 }
    si t - nacida[h] >= gracia {
      si ganMax[h] > 0 y ganHora[h] * 100 <= ganMax[h] { horasMalas[h] = horasMalas[h] + 1 } sino { horasMalas[h] = 0 }
    }
    ventasHora[h] = 0
    ingresosHora[h] = 0
    salariosHora[h] = 0
  }
}
# 6. recontar plantilla y desempleo (un hilo por agente)
fn recontar(h: entero, empleo, trab, paro, nAg: entero, nEmp: entero) {
  fijo k = h // nAg
  fijo e = empleo[h]
  si e >= 0 { sumarAtomico(trab, k * nEmp + e, 1) } sino { sumarAtomico(paro, k, 1) }
}
fn aCero(h: entero, a) { a[h] = 0 }
# 7. contratar: el desempleado busca una empresa activa con caja de sobra
fn contratar(h: entero, empleo, activa, caja, trab, t: entero, s: entero, nAg: entero, nEmp: entero, salario: entero) {
  si empleo[h] < 0 {
    fijo k = h // nAg
    var intento = 0
    mientras intento < 4 y empleo[h] < 0 {
      fijo e: entero = entero(piso(azarHilo(s + k, h, 60 + intento + t % 13) * nEmp))
      fijo ke = k * nEmp + e
      si activa[ke] == 1 y caja[ke] > (trab[ke] + 1) * salario * 4 {
        empleo[h] = e
        sumarAtomico(trab, ke, 1)
      }
      intento = intento + 1
    }
  }
}
# 8. dividendos cada 10 min: la mitad de la ganancia de la hora, por acción en circulación
fn dividendos(h: entero, activa, caja, ganHora, tenencia, trab, divPorAcc, nEmp: entero, ten: entero, acc: entero, salario: entero) {
  divPorAcc[h] = 0
  si activa[h] == 1 {
    fijo circulan = acc - tenencia[h * ten]
    fijo reserva = trab[h] * salario * 2
    var reparto = ganHora[h] // 12          # la ganancia por hora, prorrateada a 10 min, mitad
    si reparto > caja[h] - reserva { reparto = caja[h] - reserva }
    si reparto > 0 y circulan > 0 {
      fijo porAcc = reparto // circulan
      si porAcc > 0 {
        divPorAcc[h] = porAcc
        caja[h] = caja[h] - porAcc * circulan
      }
    }
  }
}
fn cobrarDividendos(h: entero, saldoInv, tenencia, divPorAcc, nInv: entero, nEmp: entero, ten: entero) {
  fijo k = h // nInv
  fijo q = h % nInv
  var total = 0
  para e en rango(nEmp) {
    fijo ke = k * nEmp + e
    total = total + tenencia[ke * ten + 1 + q] * divPorAcc[ke]
  }
  saldoInv[h] = saldoInv[h] + total
}
# 9. quiebra: la empresa con 2 horas malas seguidas cierra. Despide (los agentes
#    se enteran en «despedir»), destruye inventario, reparte la caja por acción y
#    cancela las acciones. Un hilo por empresa; los inversores cobran después.
fn quebrar(h: entero, activa, horasMalas, inv, destruido, caja, tenencia, divPorAcc, precioAcc, trab, quiebras, accVivas, nEmp: entero, ten: entero, acc: entero) {
  divPorAcc[h] = 0
  si activa[h] == 1 y horasMalas[h] >= 2 {
    fijo k = h // nEmp
    activa[h] = 0
    sumarAtomico(destruido, k, inv[h])
    inv[h] = 0
    fijo circulan = acc - tenencia[h * ten]
    si circulan > 0 {
      divPorAcc[h] = caja[h] // circulan           # liquidación: todo lo que hay
      caja[h] = caja[h] - divPorAcc[h] * circulan
    }
    # el residuo se destruye? NO: el dinero no se destruye. Se queda en la caja de
    # la empresa cerrada, que sigue siendo una cuenta (y se cuenta en el total).
    precioAcc[h] = 0
    sumarAtomico(quiebras, k, 1)
    sumarAtomico(accVivas, k, 0 - acc)
    trab[h] = 0
  }
}
fn cancelarAcciones(h: entero, tenencia, activa, precioAcc, ten: entero) {
  # h recorre economías × empresas × tenedores
  fijo ke = h // ten
  si activa[ke] == 0 y precioAcc[ke] == 0 { tenencia[h] = 0 }
}
fn despedir(h: entero, empleo, activa, nAg: entero, nEmp: entero) {
  fijo e = empleo[h]
  si e >= 0 {
    si activa[(h // nAg) * nEmp + e] == 0 { empleo[h] = -1 }
  }
}
# 10. fundar: en cada economía, los N_FUNDADORES inversores más ricos juntan hasta la
#     mitad de su dinero cada uno; si entre todos llegan al capital, fundan una
#     empresa en el sector con más ganancia (si ese sector gana) y se reparten las
#     acciones a prorrata de lo aportado. Un hilo por economía.
fn fundar(k: entero, saldoInv, activa, sector, ganHora, caja, precio, inv, trab, precioAcc, calidad, estrellas, resenas, nacida, horasMalas, ganMax, tenencia, fundadas, accVivas, aporte, t: entero, s: entero, nInv: entero, nEmp: entero, nIni: entero, nSector: entero, ten: entero, acc: entero, fundarCon: entero, nFund: entero) {
  # ¿cuánto juntan los nFund más ricos? (selección por rondas: el más rico aún no elegido)
  var juntan = 0
  para q en rango(nInv) { aporte[k * nInv + q] = 0 }
  para ronda en rango(nFund) {
    var rico = -1
    var masRico = 0
    para q en rango(nInv) {
      fijo hq = k * nInv + q
      si aporte[hq] == 0 y saldoInv[hq] > masRico {
        masRico = saldoInv[hq]
        rico = hq
      }
    }
    si rico >= 0 {
      aporte[rico] = masRico // 2
      juntan = juntan + masRico // 2
    }
  }
  # «voluntad»: aun con el dinero, solo una de cada tres horas se animan
  fijo voluntad = azarHilo(s + k, 4000 + t // 3600, 71)
  si juntan >= fundarCon y voluntad < 0.34 {
    # un hueco
    var hueco = -1
    para e en rango(nEmp) {
      fijo ke = k * nEmp + e
      si hueco < 0 y e >= nIni y activa[ke] == 0 y nacida[ke] == 0 y precioAcc[ke] == 0 { hueco = ke }
    }
    # el sector con más ganancia media por empresa activa (los vacíos atraen)
    var mejorS = -1
    var mejorG = 0
    para sec en rango(nSector) {
      var g = 0
      var n = 0
      para e en rango(nEmp) {
        fijo ke = k * nEmp + e
        si activa[ke] == 1 y sector[ke] == sec {
          g = g + ganHora[ke]
          n = n + 1
        }
      }
      var media = 1000000000
      si n > 0 { media = g // n }
      si media > mejorG {
        mejorG = media
        mejorS = sec
      }
    }
    si hueco >= 0 y mejorS >= 0 {
      # cada fundador aporta a prorrata hasta completar el capital, y recibe acciones a prorrata
      var puesto = 0
      var repartidas = 0
      para q en rango(nInv) {
        fijo hq = k * nInv + q
        si aporte[hq] > 0 {
          fijo ap: entero = aporte[hq]
          var pone: entero = ap * fundarCon // juntan
          si puesto + pone > fundarCon { pone = fundarCon - puesto }
          saldoInv[hq] = saldoInv[hq] - pone
          puesto = puesto + pone
          fijo recibe: entero = pone * acc // fundarCon
          tenencia[hueco * ten + 1 + q] = recibe
          repartidas = repartidas + recibe
          aporte[hq] = pone
        }
      }
      # el residuo de la división lo pone y lo recibe el primer fundador
      para q en rango(nInv) {
        fijo hq = k * nInv + q
        si aporte[hq] > 0 y puesto < fundarCon {
          saldoInv[hq] = saldoInv[hq] - (fundarCon - puesto)
          puesto = fundarCon
        }
      }
      tenencia[hueco * ten] = acc - repartidas      # las acciones sin dueño quedan en tesorería
      activa[hueco] = 1
      sector[hueco] = mejorS
      caja[hueco] = fundarCon
      var base = 800
      si mejorS == 1 { base = 1200 }
      si mejorS == 2 { base = 3000 }
      si mejorS == 3 { base = 1500 }
      si mejorS == 4 { base = 1000 }
      precio[hueco] = base
      inv[hueco] = 0
      trab[hueco] = 0
      precioAcc[hueco] = fundarCon // acc
      calidad[hueco] = 1 + entero(piso(azarHilo(s + k, 3000 + t, 70) * 5))
      estrellas[hueco] = 5 * 300
      resenas[hueco] = 5
      nacida[hueco] = t
      horasMalas[hueco] = 0
      ganMax[hueco] = 0
      fundadas[k] = fundadas[k] + 1
      accVivas[k] = accVivas[k] + acc
    }
  }
}
# 11. bolsa: subasta de precio único por empresa, UN HILO POR EMPRESA (secuencial
#     dentro, paralela entre empresas y economías). Las 48 empresas de una misma
#     economía tocan a los MISMOS inversores a la vez, así que el dinero de los
#     inversores se mueve con sumarAtomico; con «=» dos empresas se pisaban y el
#     dinero se perdía o se duplicaba (salió con 100 economías: −93 monedas en
#     una y +190 en otra). Las acciones de cada empresa son solo de su hilo. Los inversores de la economía
#     deciden lado y cantidad; el precio se mueve con el desequilibrio y las
#     acciones cambian de mano al precio de cierre, tesorería incluida.
fn bolsa(h: entero, activa, precioAcc, precioAccH, tenencia, saldoInv, divPorAcc, ganHora, caja, trab, t: entero, s: entero, nEmp: entero, nInv: entero, ten: entero, acc: entero, salario: entero) {
  si activa[h] == 1 {
    fijo k = h // nEmp
    fijo p: entero = precioAcc[h]
    fijo pH: entero = precioAccH[h]
    # valor fundamental: caja por acción, menos los salarios que debe esta hora,
    # más 40 horas de la ganancia actual
    fijo cj: entero = caja[h]
    fijo gh: entero = ganHora[h]
    fijo tr: entero = trab[h]
    var valor: entero = (cj - tr * salario + gh * 40) // acc
    si valor < 1 { valor = 1 }
    var compras = 0
    var ventas = 0
    fijo tes: entero = tenencia[h * ten]
    var ofertaTes: entero = 0
    para q en rango(nInv) {
      fijo hq = k * nInv + q
      fijo r = azarHilo(s + k, 5000 + h % nEmp + 97 * q, 80 + t % 17)
      fijo r2 = azarHilo(s + k, 5000 + h % nEmp + 97 * q, 100 + t % 17)
      fijo estilo = q % 3
      var lado = -1
      si estilo == 0 {
        si p * 10 < valor * 9 { lado = 0 } sino si p * 10 > valor * 11 { lado = 1 }
      } sino si estilo == 1 {
        # sigue la ganancia: compra si la empresa gana, vende si pierde, y sin
        # datos (la primera hora) no hace nada
        # sigue la tendencia del precio: compra si subió más de un 2 % desde el
        # cierre de la hora anterior, vende si bajó más de un 2 %; actúa 1 de
        # cada 6 sesiones. Son los que inflan burbujas y las pinchan.
        si r2 < 0.17 y pH > 0 {
          si p * 100 > pH * 102 { lado = 0 } sino si p * 100 < pH * 98 { lado = 1 }
        }
      } sino {
        si r < 0.5 { lado = 0 } sino { lado = 1 }
      }
      si lado == 0 {
        fijo dinero: entero = saldoInv[hq]
        var pp = p
        si pp < 1 { pp = 1 }
        fijo quiere: entero = (dinero * (1 + entero(piso(r * 5))) // 100) // pp
        compras = compras + quiere
      } sino si lado == 1 {
        fijo tengo: entero = tenencia[h * ten + 1 + q]
        ventas = ventas + tengo * (1 + entero(piso(r * 5))) // 100
      }
    }
    # la tesorería sale a vender solo cuando hay más compras que ventas, y cubre
    # como mucho la mitad del exceso: así la emisión frena las subidas sin hundir
    # el precio cuando nadie compra
    si compras > ventas {
      ofertaTes = (compras - ventas) // 2
      si ofertaTes > tes // 50 { ofertaTes = tes // 50 }
      si ofertaTes > tes { ofertaTes = tes }
      ventas = ventas + ofertaTes
    }
    # precio nuevo según el desequilibrio, acotado al ±0,5 % por sesión (60 sesiones
    # por hora: hasta ±35 % en una hora si todos empujan al mismo lado)
    var pNuevo = p
    si compras + ventas > 0 {
      fijo deseq = (compras - ventas) * 100 // (compras + ventas)   # -100..100
      pNuevo = p + p * deseq // 20000
      si pNuevo < 1 { pNuevo = 1 }
    }
    var cruzan = compras
    si ventas < cruzan { cruzan = ventas }
    si cruzan > 0 {
      # vendedores: tesorería primero, luego inversores en proporción; los
      # compradores reparten lo cruzado en proporción a lo que querían
      var quedan = cruzan
      var deTes = ofertaTes
      si quedan < deTes { deTes = quedan }
      tenencia[h * ten] = tenencia[h * ten] - deTes
      caja[h] = caja[h] + deTes * pNuevo
      quedan = quedan - deTes
      para q en rango(nInv) {
        fijo hq = k * nInv + q
        fijo r = azarHilo(s + k, 5000 + h % nEmp + 97 * q, 80 + t % 17)
        fijo r2 = azarHilo(s + k, 5000 + h % nEmp + 97 * q, 100 + t % 17)
        fijo estilo = q % 3
        var vende = 0
        si estilo == 0 y p * 10 > valor * 11 { vende = 1 }
        si estilo == 1 y r2 < 0.17 y pH > 0 y p * 100 < pH * 98 { vende = 1 }
        si estilo == 2 y r >= 0.5 { vende = 1 }
        si vende == 1 y quedan > 0 {
          fijo tengo: entero = tenencia[h * ten + 1 + q]
          var v: entero = tengo * (1 + entero(piso(r * 5))) // 100
          si v > quedan { v = quedan }
          tenencia[h * ten + 1 + q] = tengo - v
          sumarAtomico(saldoInv, hq, v * pNuevo)
          quedan = quedan - v
        }
      }
      fijo vendidas = cruzan - quedan
      var porRepartir = vendidas
      para q en rango(nInv) {
        fijo hq = k * nInv + q
        fijo r = azarHilo(s + k, 5000 + h % nEmp + 97 * q, 80 + t % 17)
        fijo r2 = azarHilo(s + k, 5000 + h % nEmp + 97 * q, 100 + t % 17)
        fijo estilo = q % 3
        var compra = 0
        si estilo == 0 y p * 10 < valor * 9 { compra = 1 }
        si estilo == 1 y r2 < 0.17 y pH > 0 y p * 100 > pH * 102 { compra = 1 }
        si estilo == 2 y r < 0.5 { compra = 1 }
        si compra == 1 y porRepartir > 0 {
          var pp = p
          si pp < 1 { pp = 1 }
          fijo dinero: entero = saldoInv[hq]
          fijo quiere: entero = (dinero * (1 + entero(piso(r * 5))) // 100) // pp
          var cmp = compras
          si cmp < 1 { cmp = 1 }
          var c = quiere * vendidas // cmp
          si c > porRepartir { c = porRepartir }
          fijo tiene: entero = saldoInv[hq]
          si c * pNuevo > tiene { c = tiene // pNuevo }
          tenencia[h * ten + 1 + q] = tenencia[h * ten + 1 + q] + c
          sumarAtomico(saldoInv, hq, 0 - c * pNuevo)
          porRepartir = porRepartir - c
        }
      }
      # lo que nadie pudo comprar vuelve a la tesorería: ninguna acción se pierde
      si porRepartir > 0 {
        tenencia[h * ten] = tenencia[h * ten] + porRepartir
        caja[h] = caja[h] - porRepartir * pNuevo
        # … y el dinero que la tesorería había cobrado de más vuelve a quien no recibió.
        # Como esos compradores ya no se identifican, lo cobra la tesorería a cambio de
        # quedarse las acciones: la cuenta cuadra porque arriba se restó el dinero solo
        # a quien SÍ recibió. (El ajuste de caja de arriba corrige el deTes de más.)
        si caja[h] < 0 { caja[h] = 0 }
      }
    }
    precioAcc[h] = pNuevo
  }
}
# 12. verificación por economía: suma de dinero, acciones vivas, bienes
fn sumarDineroAg(h: entero, saldo, sumaDinero, nAg: entero) { sumarAtomico(sumaDinero, h // nAg, saldo[h]) }
fn sumarDineroInv(h: entero, saldoInv, sumaDinero, nInv: entero) { sumarAtomico(sumaDinero, h // nInv, saldoInv[h]) }
fn sumarDineroEmp(h: entero, caja, sumaDinero, nEmp: entero) { sumarAtomico(sumaDinero, h // nEmp, caja[h]) }
fn sumarTenencia(h: entero, tenencia, sumaAcc, ten: entero) { sumarAtomico(sumaAcc, h // ten, tenencia[h]) }
fn verificar(k: entero, sumaDinero, dinero0, sumaAcc, activa, accVivas, inv, producido, consumido, destruido, fallo, nEmp: entero, acc: entero) {
  si sumaDinero[k] != dinero0[k] { fallo[k] = 2 }
  var vivas = 0
  var enEstante = 0
  para e en rango(nEmp) {
    fijo ke = k * nEmp + e
    si activa[ke] == 1 {
      si sumaAcc[ke] != acc { fallo[k] = 4 }
      vivas = vivas + acc
    } sino {
      si sumaAcc[ke] != 0 { fallo[k] = 4 }
    }
    enEstante = enEstante + inv[ke]
  }
  si vivas != accVivas[k] { fallo[k] = 4 }
  si producido[k] != consumido[k] + enEstante + destruido[k] { fallo[k] = 5 }
}
fn medir(k: entero, activa, precioAcc, indiceBolsa, indiceHace1h, caidaMax, paro, paroMax, nEmp: entero) {
  var suma = 0
  var n = 0
  para e en rango(nEmp) {
    fijo ke = k * nEmp + e
    si activa[ke] == 1 {
      suma = suma + precioAcc[ke]
      n = n + 1
    }
  }
  si n > 0 { indiceBolsa[k] = suma // n }
  si paro[k] > paroMax[k] { paroMax[k] = paro[k] }
  si indiceHace1h[k] > 0 {
    fijo caida = (indiceHace1h[k] - indiceBolsa[k]) * 100 // indiceHace1h[k]
    si caida > caidaMax[k] { caidaMax[k] = caida }
  }
}
fn guardarIndice(k: entero, indiceBolsa, indiceHace1h) { indiceHace1h[k] = indiceBolsa[k] }
fn guardarPrecioAcc(h: entero, precioAcc, precioAccH) { precioAccH[h] = precioAcc[h] }
