# ═══ Monitoreo: el promedio esconde justo lo que se está buscando ═══════════
# Un tablero de sensores que enseña «temperatura promedio: 25 °C · NORMAL»
# puede estar tapando dos invernaderos rotos: uno a 40 y otro a 10 promedian 25.
# El promedio es la estadística equivocada para vigilar, porque vigilar es
# buscar el que se salió, y promediar es exactamente la operación que lo
# disuelve. Lo que hay que mirar son los extremos, la amplitud y cada zona.
# Los datos de abajo están puestos a propósito para que el promedio global
# quede en rango mientras dos zonas están fuera.
#
# Y otras dos del original:
#   · «calcularPromedioTemperatura» y «calcularPromedioHumedad» eran dos
#     funciones de quince líneas cada una haciendo lo que hace «media».
#   · había un indicador llamado «eficiencia» que era 100 menos el porcentaje
#     de lecturas fuera de rango. Eso no es la eficiencia de nada: es el
#     porcentaje de lecturas en rango, y llamarlo eficiencia hace que alguien
#     lo lleve a una reunión como si midiera la producción.
# núcleo: web · eDSL: numerico, datos, formato, estilo

usar media, mediana, desviacion, suma de "numerico"
usar deRegistros, conColumna, ordenarPor, columna, donde, aRegistros de "datos"
usar decimales, porciento, plural, enNegrita, regla, enRojo, enVerde,
     enAmarillo, enGris, alinearDer, alinearIzq, tablaTexto,
     minigrafico de "formato"
usar tema, estiloGlobal, reiniciar, estilo, color, espacio, radio, tipografia,
     paleta, escalaTipo, enOscuro de "estilo"

fijo TEMP_MAX: real = 28.0
fijo HUM_MIN: real = 55.0
fijo HUM_MAX: real = 80.0

# ── el estado, calculado UNA vez por lectura ──────────────────────────────
fn estadoDe(t: real, h: real): texto {
    si t < TEMP_MIN - 5.0 o t > TEMP_MAX + 5.0 { devolver "1 crítico" }
    si t < TEMP_MIN o t > TEMP_MAX { devolver "2 alerta" }
    si h < HUM_MIN o h > HUM_MAX { devolver "2 alerta" }
    devolver "3 en rango"
}

var t = deRegistros(lecturas)
t = conColumna(t, "estado", fn(f: dic): texto {
    devolver estadoDe(f["temp"], f["hum"])
})

fijo temps: lista = columna(t, "temp")
fijo hums: lista = columna(t, "hum")
fijo zonas: lista = ["Invernadero A", "Invernadero B", "Silo Secado", "Cuarto Frío"]

fn deZona(z: texto): lista {
    devolver aRegistros(donde(t, fn(f: dic): bool { devolver f["zona"] == z }))
}
fn cuantas(e: texto): entero {
    devolver longitud(columna(donde(t, fn(f: dic): bool {
        devolver f["estado"] == e
    }), "zona"))
}
fijo enRango: entero = cuantas("3 en rango")

# ── la página ─────────────────────────────────────────────────────────────
fn fichaDe(rot: texto, c: texto, det: texto): nodo {
    devolver etiqueta("div", ficha,
        etiqueta("div", rotulo, rot),
        etiqueta("div", cifra, c),
        etiqueta("div", detalle, det))
}

fn selloZona(z: texto): nodo {
    fijo filas: lista = deZona(z)
    var peor: texto = "3 en rango"
    para f en filas { si f["estado"] < peor { peor = f["estado"] } }
    si peor == "1 crítico" {
        devolver etiqueta("span", estilo(sello, {"fondo": color("mal")}), "crítico")
    }
    si peor == "2 alerta" {
        devolver etiqueta("span", estilo(sello, {"fondo": color("ojo")}), "alerta")
    }
    devolver etiqueta("span", estilo(sello, {"fondo": color("bien")}), "en rango")
}

fn filaZona(z: texto): nodo {
    fijo filas: lista = deZona(z)
    fijo ts: lista = mapear(filas, fn(f: dic): real { devolver f["temp"] })
    devolver etiqueta("div", filaZ,
        etiqueta("span", estilo({"flex": "1 1 auto", "grosor": 600}), z),
        etiqueta("span", estilo(detalle, {"minAncho": 110}),
                 decimales(minimo(ts), 1) + " – " + decimales(maximo(ts), 1) + " °C"),
        etiqueta("span", estilo(detalle, {"minAncho": 90}),
                 "media " + decimales(media(ts), 1)),
        selloZona(z))
}

        "calculado: está mal elegido, porque promediar es justo la operación " +

                    decimales(maximo(temps), 1)),
            fichaDe("dispersión", decimales(desviacion(temps), 1) + " °C",
                    "con una media de " + decimales(media(temps), 1)),
            fichaDe("fuera de rango", texto(longitud(lecturas) - enRango),
                    "de " + texto(longitud(lecturas)) + " lecturas"),
            fichaDe("lecturas en rango", porciento(real(enRango) /
                    real(longitud(lecturas))),
                    "no es «eficiencia»: es esto y nada más"))),

    etiqueta("div", panel,
        etiqueta("h2", h2, "Zona por zona"),
        mapear(zonas, filaZona))
)))

fn nombreEstado(e: texto): texto {
    si e == "1 crítico" { devolver "CRÍTICO" }
    si e == "2 alerta" { devolver "alerta" }
    devolver "en rango"
}
